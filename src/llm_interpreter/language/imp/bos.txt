ℤ := Set of integers
𝔹 := {{true, false}} , set of booleans
𝜎 := {{ ID ↦ ℤ }}
v := integer value
x := integer variable

q := boolean value

S := statement

configuration := 〈operation,𝜎〉
big-step transition := 〈e,𝜎〉 ⇓ 〈v,𝜎〉 , e evaluates to v in one big-step
empty computation := 〈ε,𝜎〉
halt execution := 〈HALT,𝜎〉, halt program execution
