module {name}-SYNTAX
  imports INT
  imports BOOL
  imports LIST
  imports MAP
  imports ID
  imports STRING
  imports K-LOCATIONS

  syntax AExp  ::= Int | Id
                 | "{PLUS_OP}" AExp                   [strict]
                 | "{MINUS_OP}" AExp                   [strict]
                 > left:
                   AExp "{DIV_OP}" AExp              [strict]
                 | AExp "{MUL_OP}" AExp              [strict]
                 | AExp "{MOD_OP}" AExp              [strict]
                 > left:
                   AExp "{PLUS_OP}" AExp              [strict]
                 | AExp "{MINUS_OP}" AExp              [strict]
                 > PAExp
  syntax PAExp ::= "(" AExp ")"                      [klabel(parenthesizedA), symbol]
  syntax BExp  ::= Bool
                 > "{NOT_OP}" BExp                   [strict]
                 > left:
                   AExp "{LT_OP}" AExp               [seqstrict]
                 | AExp "{LTEQ_OP}" AExp             [seqstrict]
                 | AExp "{GT_OP}" AExp               [seqstrict]
                 | AExp "{GTEQ_OP}" AExp             [seqstrict]
                 | AExp "{EQ_OP}" AExp               [seqstrict]
                 | AExp "{NEQ_OP}" AExp              [seqstrict]
                 > left:
                   BExp "{AND_OP}" BExp              [seqstrict]
                 | BExp "{OR_OP}" BExp               [seqstrict]
                 > PBExp
  syntax PBExp ::= "(" BExp ")"                      [klabel(parenthesizedB), symbol]

  syntax Block      ::= "{{" "}}"
                      | "{{" Stmt "}}"
  syntax ContMark   ::= "continueMarker"
  syntax WhileStmt  ::= "{WHILE}" "(" BExp ")" Block ";"
  syntax CompStmt   ::= Id "{ASSIGN_OP}" AExp ";"                     [strict(2)]
                      | "{IF}" "(" BExp ")" Block "{ELSE}" Block ";"  [strict(1)]
                      | WhileStmt
                      | "{WHILE}1" "(" BExp ")" Block
                      | "{HALT}" ";"
                      | "{BREAK}" ";"
                      | "{CONTINUE}" ";"
                      | "int" Id ";"
  syntax Stmt       ::= Block
                      | CompStmt
                      | ContMark
                      | "Empty"
                      > Stmt Stmt                         [left]

  syntax WhileStmt  [locations]
  syntax CompStmt   [locations]

  syntax Pgm ::= Stmt
endmodule
